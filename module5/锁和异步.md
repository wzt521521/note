## 一、锁相关核心代码

### 1. std::mutex 基础互斥锁

```
#include <mutex>
std::mutex mtx;
int cnt = 0;

void add() {
    mtx.lock();       // 阻塞加锁，独占，不可重入（同线程二次lock直接死锁）
    cnt++;
    mtx.unlock();     // 必须配对释放，否则死锁

    // 非阻塞版本：拿到锁返回true，否则false
    if (mtx.try_lock()) {
        cnt++;
        mtx.unlock();
    }
}
```

### 2. lock_guard vs unique_lock（高频对比）

```
#include <mutex>
std::mutex mtx;

void test_lock_guard() {
    // RAII自动加解锁，仅作用域管理，不可手动解锁、不可移动，开销最小
    std::lock_guard<std::mutex> lg(mtx); 
    // 离开作用域自动unlock
}

void test_unique_lock() {
    // 1. 支持延迟加锁
    std::unique_lock<std::mutex> ul(mtx, std::defer_lock); 
    ul.lock();         // 手动加锁
    ul.unlock();       // 手动解锁（可提前释放锁）
    // 2. 支持超时尝试、支持移动语义
    // 3. 条件变量必须用unique_lock（wait内部要释放/重新加锁）
}
```

### 3. std::scoped_lock（C++17 多锁防死锁）

```
#include <mutex>
std::mutex m1, m2;

void safe_lock() {
    // 内部按固定顺序同时加多个锁，天然避免循环等待死锁
    // 等价于 std::lock(m1,m2) + 多个lock_guard，写法更简洁
    std::scoped_lock sl(m1, m2); 
    // 离开作用域按逆序自动解锁
}
```

### 4. std::shared_mutex 读写锁（读共享、写独占）

```
#include <shared_mutex>
std::shared_mutex smtx;
int data = 0;

void read() {
    // 读操作：共享锁，多个读线程可同时进入
    std::shared_lock<std::shared_mutex> sl(smtx);
    int val = data; // 只读
}

void write() {
    // 写操作：独占锁，阻塞所有读写
    std::unique_lock<std::shared_mutex> ul(smtx);
    data = 100;
}
```

### 5. 自旋锁（atomic_flag 实现）

```
#include <atomic>
std::atomic_flag flag = ATOMIC_FLAG_INIT;

void spin_lock() {
    // test_and_set：置为true并返回旧值
    // 旧值为true说明锁被占，循环忙等（不进入内核睡眠，无上下文切换）
    while (flag.test_and_set(std::memory_order_acquire)); 
}

void spin_unlock() {
    flag.clear(std::memory_order_release);
}
// 适用：临界区极短、多核CPU；缺点：长时间等待空转耗CPU
```

### 6. 死锁典型场景与避免

```
// ❌ 死锁：线程1拿m1等m2，线程2拿m2等m1（循环等待）
void dead_lock() {
    // 线程1
    m1.lock(); m2.lock(); 
    // 线程2
    m2.lock(); m1.lock(); 
}

// ✅ 避免：固定顺序加锁 / 用scoped_lock一次性加锁
void safe_lock() {
    std::scoped_lock sl(m1, m2); // 推荐，内部自动处理顺序
}
```

## 二、异步编程核心代码

### 1. std::thread 基础

```
#include <thread>
#include <iostream>

void func(int a) { std::cout << a; }

int main() {
    std::thread t(func, 10); // 创建线程，立即执行
    t.join();                // 阻塞等待线程结束，回收资源
    // t.detach();           // 分离线程，脱离主线程管理，生命周期自主
    // 注意：thread对象销毁前必须join或detach，否则程序直接terminate
    return 0;
}
```

### 2. promise / future 一对一结果传递

```
#include <future>
#include <thread>

void producer(std::promise<int> p) {
    p.set_value(100); // 生产者：设置结果（只能set一次）
    // p.set_exception(std::make_exception_ptr(std::runtime_error("err")));
}

int main() {
    std::promise<int> p;
    std::future<int> f = p.get_future(); // 消费者：关联future

    std::thread t(producer, std::move(p));
    int res = f.get(); // 阻塞等待结果，只能调用一次
    t.join();
    return 0;
}
```

### 3. std::packaged_task 任务封装

```
#include <future>
#include <thread>

int add(int a, int b) { return a + b; }

int main() {
    // 包装可调用对象，自动绑定future
    std::packaged_task<int(int, int)> task(add);
    std::future<int> f = task.get_future();

    std::thread t(std::move(task), 1, 2); // 任务执行完自动set_value
    int res = f.get();
    t.join();
    return 0;
}
// 与promise区别：侧重封装"可执行任务"，无需手动set_value
```

### 4. std::async 高层异步接口

```
#include <future>

int task() { return 100; }

int main() {
    // 强制异步：立即创建新线程执行
    std::future<int> f1 = std::async(std::launch::async, task);

    // 延迟执行：不创建线程，调用get/wait时才在当前线程同步执行（懒执行）
    std::future<int> f2 = std::async(std::launch::deferred, task);

    // 默认策略：async | deferred，由编译器实现自主选择
    std::future<int> f3 = std::async(task);

    f1.get();
    return 0;
}
```

### 5. condition_variable 条件变量（生产者消费者）

```
#include <mutex>
#include <condition_variable>
#include <queue>

std::mutex mtx;
std::condition_variable cv;
std::queue<int> q;

void producer() {
    std::unique_lock<std::mutex> ul(mtx);
    q.push(1);
    cv.notify_one(); // 唤醒一个等待线程
}

void consumer() {
    std::unique_lock<std::mutex> ul(mtx);
    // 必须用while，不能用if —— 防止虚假唤醒（无notify也可能被唤醒）
    // wait内部：释放锁 → 阻塞 → 被唤醒 → 重新加锁 → 检查谓词
    while (q.empty()) {
        cv.wait(ul); 
    }
    int val = q.front(); q.pop();
}
// 为什么配unique_lock？因为wait要中途释放锁、唤醒后重新加锁，lock_guard不支持
```

### 6. std::atomic 原子操作与内存序

```
#include <atomic>

std::atomic<int> cnt{0};
std::atomic<bool> ready{false};

void add() {
    cnt.fetch_add(1, std::memory_order_relaxed); // 原子自增，松散序
}

// acquire-release 语义：保证写可见性
void writer() {
    cnt.store(100, std::memory_order_release); // 写：release
    ready.store(true, std::memory_order_release);
}

void reader() {
    while (!ready.load(std::memory_order_acquire)); // 读：acquire
    // 此时一定能看到cnt=100
    int val = cnt.load(std::memory_order_acquire);
}
// 默认内存序 memory_order_seq_cst：顺序一致性，最强，开销最大
```

### 7. 极简线程池

```
#include <thread>
#include <queue>
#include <functional>
#include <mutex>
#include <condition_variable>

class ThreadPool {
public:
    ThreadPool(size_t n) : stop(false) {
        for (size_t i = 0; i < n; ++i) {
            workers.emplace_back([this] {
                while (true) {
                    std::function<void()> task;
                    {
                        std::unique_lock<std::mutex> ul(mtx);
                        cv.wait(ul, [this] { return stop || !tasks.empty(); });
                        if (stop && tasks.empty()) return;
                        task = std::move(tasks.front()); tasks.pop();
                    }
                    task(); // 执行任务
                }
            });
        }
    }
    // 提交任务
    template<class F>
    void enqueue(F&& f) {
        std::lock_guard<std::mutex> lg(mtx);
        tasks.emplace(std::forward<F>(f));
        cv.notify_one();
    }
    ~ThreadPool() {
        { std::lock_guard<std::mutex> lg(mtx); stop = true; }
        cv.notify_all();
        for (auto& t : workers) t.join();
    }
private:
    std::vector<std::thread> workers;   // 工作线程组
    std::queue<std::function<void()>> tasks; // 任务队列
    std::mutex mtx;
    std::condition_variable cv;
    bool stop;
};
```
