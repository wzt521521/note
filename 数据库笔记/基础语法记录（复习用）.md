CREATE USER 'reactor_app'@'127.0.0.1'
    IDENTIFIED BY '123123';

GRANT SELECT, INSERT, UPDATE, DELETE
ON reactor_lab.*
TO 'reactor_app'@'127.0.0.1';

SHOW DATABASES;
-- 等价
SHOW SCHEMAS;


CREATE TABLE permission_probe (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    note VARCHAR(100) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id)
) ENGINE = InnoDB;

USE reactor_lab;

-- 清理权限实验留下的临时表
DROP TABLE IF EXISTS permission_probe;

-- 为了让脚本可以重复执行，先删除子表，再删除父表
DROP TABLE IF EXISTS employees;
DROP TABLE IF EXISTS departments;

CREATE TABLE departments (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    created_at DATETIME(6) NOT NULL
        DEFAULT CURRENT_TIMESTAMP(6),

    PRIMARY KEY (id),
    UNIQUE KEY uk_departments_name (name)
) ENGINE = InnoDB;

CREATE TABLE employees (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    department_id BIGINT UNSIGNED NOT NULL,
    employee_no VARCHAR(20) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(254) NULL,
    salary DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    hired_on DATE NOT NULL,
    status TINYINT UNSIGNED NOT NULL DEFAULT 1,
    created_at DATETIME(6) NOT NULL
        DEFAULT CURRENT_TIMESTAMP(6),
    updated_at DATETIME(6) NOT NULL
        DEFAULT CURRENT_TIMESTAMP(6)
        ON UPDATE CURRENT_TIMESTAMP(6),

    PRIMARY KEY (id),
    UNIQUE KEY uk_employees_employee_no (employee_no),
    UNIQUE KEY uk_employees_email (email),

    CONSTRAINT chk_employees_salary
        CHECK (salary >= 0),

    CONSTRAINT chk_employees_status
        CHECK (status IN (0, 1)),

    CONSTRAINT fk_employees_department
        FOREIGN KEY (department_id)
        REFERENCES departments(id)
) ENGINE = InnoDB;
